NUM_ROW EQU 3
NUM_COL EQU 4

				AREA initialMap, DATA, READONLY			
initial_maze	DCB 0, 1, 2, 3
		DCB 4, 5, 6, 7					
		DCB 8, 9, 10, 11	
initial_horizontal_walls DCB 1, 1, 1, 2, 1, 1, 1, 2, 1, 1, 1, 2
initial_vertical_walls   DCB 1, 1, 1, 1, 1, 1, 1, 1, 2, 2, 2, 2

 				AREA currentMap, DATA, READWRITE
maze			SPACE NUM_ROW * NUM_COL
horizontal_walls	SPACE NUM_ROW * NUM_COL
vertical_walls	SPACE NUM_ROW * NUM_COL
	
	

                AREA    |.text|, CODE, READONLY


; Reset Handler

Reset_Handler   PROC
                EXPORT  Reset_Handler             [WEAK]
                ;IMPORT  SystemInit
                ;IMPORT  __main
                ;LDR     R0, =SystemInit
                ;BLX     R0
                ;LDR     R0, =__main
                ;BX      R0
				
				; initialize the data structure
				LDR r0, =initial_maze
				LDR r1, =maze
				LDR r2, =initial_horizontal_walls
				LDR r3, =horizontal_walls
				LDR r4, =initial_vertical_walls
				LDR r5, =vertical_walls
				MOV r6, #NUM_ROW * NUM_COL
loopCopyData	LDRB r7, [r0], #1
				STRB r7, [r1], #1
				LDRB r7, [r2], #1
				STRB r7, [r3], #1
				LDRB r7, [r4], #1
				STRB r7, [r5], #1
				SUBS r6, r6, #1
				BNE loopCopyData
				
				LDR r0, =maze
				LDR r1, =horizontal_walls
				LDR r2, =vertical_walls
				MOV r3, #NUM_ROW
				MOV r4, #NUM_COL
				MOV r5, #4	; increment
				MOV r6, #2	; offset
				PUSH {r4-r6}
				BL kruskal
				POP {r4-r6}
stop			B stop
                ENDP

ADD_MAZ			RN	0
ADD_HL			RN	1
ADD_VL			RN	2
ROW				RN	3
COL				RN	4
Y				RN	5
X				RN	6
ROWCOL			RN	7
TEMP			RN	8
TEMP2			RN	9
M				RN	10
N				RN	11
TEMP3			RN	12

kruskal			PROC
				PUSH{R7-R11,LR}
				
				
				MOV TEMP2, #0
				MOV M, #0
				MOV N, #0
				MOV ROWCOL, #0
				MOV TEMP, #0
				
WHILE
BEGIN			
				MUL ROWCOL, ROW, COL
				MOV TEMP, #0
				MOV TEMP2,#0
				MOV TEMP3, #0
WHILE2
				LDRB TEMP, [ADD_MAZ, TEMP2]
				ADD TEMP3, TEMP
				SUB ROWCOL, #1
				ADD TEMP2, #1
				CMP ROWCOL, #0
				BNE WHILE2
				CMP TEMP3, #0
				BEQ FINITO
				
				
				ADD X, X, Y
				MUL ROWCOL, ROW, COL
				CMP X, ROWCOL
				BHS ELSS2
				LDRB TEMP, [ADD_HL, X]
				CMP TEMP, #1
				BNE ELSEIF
				LDRB TEMP, [ADD_MAZ,X]
				ADD TEMP2, X, #1
				LDRB TEMP2, [ADD_MAZ, TEMP2]
				CMP TEMP, TEMP2
				BEQ BEGIN
				LDR TEMP, =0
				STRB TEMP, [ADD_HL,X]
				LDRB TEMP, [ADD_MAZ, X]
				ADD TEMP2, X, #1
				LDRB TEMP2, [ADD_MAZ, TEMP2]
				CMP TEMP, TEMP2
				BHI MAX
				MOV M, TEMP
				MOV N, TEMP2
				B CHANGE
MAX
				MOV N, TEMP
				MOV M, TEMP2
CHANGE
				MOV TEMP2, #0
CHANGE1				
				LDRB TEMP, [ADD_MAZ,TEMP2]
				CMP TEMP, N
				BNE	CON
				STRB M, [ADD_MAZ, TEMP2]
				ADD TEMP2,#1
				SUB ROWCOL, #1
				CMP ROWCOL, #0
				BEQ BEGIN
				B CHANGE1
CON
				ADD TEMP2, #1
				SUB ROWCOL, #1
				CMP ROWCOL, #0
				BEQ BEGIN
				B CHANGE1
				
ELSEIF
				CMP TEMP, #0
				BNE BEGIN
				ADD Y,Y,#1
				B BEGIN
				

	
ELSS2
				SUB X, X, ROWCOL
				CMP X, ROWCOL
				BHS BEGIN
				LDRB TEMP, [ADD_VL, X]
				CMP TEMP, #1
				BNE ELSEIF2
				LDRB TEMP, [ADD_MAZ,X]
				ADD TEMP2, X, COL
				LDRB TEMP2, [ADD_MAZ, TEMP2]
				CMP TEMP, TEMP2
				BEQ BEGIN
				LDR TEMP, =0
				STRB TEMP, [ADD_VL,X]
				LDRB TEMP, [ADD_MAZ, X]
				ADD TEMP2, X, COL
				LDRB TEMP2, [ADD_MAZ, TEMP2]
				CMP TEMP, TEMP2
				BHI MAX2
				MOV M, TEMP
				MOV N, TEMP2
				B CHANGE2
MAX2
				MOV N, TEMP
				MOV M, TEMP2				
CHANGE2
				MOV TEMP2, #0
CHANGE3
				LDRB TEMP, [ADD_MAZ,TEMP2]
				CMP TEMP, N
				BNE	CON1
				STRB M, [ADD_MAZ, TEMP2]
				ADD TEMP2,#1
				SUB ROWCOL, #1
				CMP ROWCOL, #0
				BEQ BEGIN
				B CHANGE3
CON1
				ADD TEMP2, #1
				SUB ROWCOL, #1
				CMP ROWCOL, #0
				BEQ BEGIN
				B CHANGE3
	
ELSEIF2
				CMP TEMP, #0
				BNE BEGIN
				ADD Y,Y,#1
				B BEGIN

FINITO

				
				POP{R7-R11,PC}
				ENDP
